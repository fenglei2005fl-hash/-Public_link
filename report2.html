exp_first_screen AS (
    SELECT
        dt,
        CAST(member_id AS STRING)                    AS member_id,
        '实验组'                                      AS exp_group,
        MIN(log_timestamp)                            AS first_screen_ts
    FROM dw.dw_tfc_elmt_eps_ug_di
    WHERE dt BETWEEN '20260526' AND '20260602'
      AND p_game_tp = '106'
      AND site_tp   = 'shein'
      AND game_tp   = 106
      AND page_tp   = '2'
      AND event_nm  = 'expose_popup_selectattr'
      AND get_json_object(activity_param,'$.popup_type')       = 'groupbuycountdown_begin'
      AND get_json_object(activity_param,'$.groupbuycountdown') IN ('A','B')
      AND COALESCE(get_json_object(page_param,'$.groupid'),'') = ''  -- 团长
      AND member_id IS NOT NULL AND member_id > 0
    GROUP BY dt, CAST(member_id AS STRING)
),
-- 步骤1b：对照组团长首屏首次曝光
ctrl_first_screen AS (
    SELECT dt, CAST(member_id AS STRING) AS member_id,
           '对照组' AS exp_group, MIN(log_timestamp) AS first_screen_ts
    FROM dw.dw_tfc_elmt_eps_ug_di
    WHERE dt BETWEEN '20260526' AND '20260602'
      AND p_game_tp='106' AND site_tp='shein' AND game_tp=106 AND page_tp='2'
      AND event_nm='expose_popup_selectattr'
      AND get_json_object(activity_param,'$.popup_type')='groupbuyfreeshipping'
      AND COALESCE(get_json_object(page_param,'$.groupid'),'')=''
      AND member_id IS NOT NULL AND member_id > 0
    GROUP BY dt, CAST(member_id AS STRING)
),
all_first_screen AS (
    SELECT dt,member_id,exp_group,first_screen_ts FROM exp_first_screen
    UNION ALL
    SELECT dt,member_id,exp_group,first_screen_ts FROM ctrl_first_screen
),
-- 步骤2：立即购曝光明细
orderdrawer_detail AS (
    SELECT dt, CAST(member_id AS STRING) AS member_id, log_timestamp AS orderdrawer_ts
    FROM dw.dw_tfc_elmt_eps_ug_di
    WHERE dt BETWEEN '20260526' AND '20260602'
      AND p_game_tp='106' AND site_tp='shein' AND game_tp=106 AND page_tp='2'
      AND event_nm='expose_popup_selectattr'
      AND get_json_object(activity_param,'$.popup_type')='orderdrawer'
      AND COALESCE(get_json_object(page_param,'$.groupid'),'')=''
      AND member_id IS NOT NULL AND member_id > 0
),
-- 步骤3：计算时间间隔（秒）
user_interval AS (
    SELECT f.dt, f.exp_group, f.member_id,
           CAST(MIN(o.orderdrawer_ts)-f.first_screen_ts AS DOUBLE)/1000.0 AS interval_sec
    FROM all_first_screen f
    INNER JOIN orderdrawer_detail o
        ON f.dt=o.dt AND f.member_id=o.member_id
        AND o.orderdrawer_ts > f.first_screen_ts
    GROUP BY f.dt,f.exp_group,f.member_id,f.first_screen_ts
),
-- 步骤4：NTILE(10) 按日×组分桶
user_ntile AS (
    SELECT dt,exp_group,member_id,interval_sec,
           NTILE(10) OVER(PARTITION BY dt,exp_group ORDER BY interval_sec ASC) AS ntile_bucket
    FROM user_interval
)
SELECT
    dt, exp_group,
    CONCAT(CAST(ntile_bucket*10 AS STRING),'分位')         AS percentile_group,
    ROUND(AVG(interval_sec),2)                             AS avg_interval_sec,
    ROUND(PERCENTILE_APPROX(interval_sec,0.5),2)           AS median_interval_sec,
    ROUND(MIN(interval_sec),2)                             AS min_interval_sec,
    ROUND(MAX(interval_sec),2)                             AS max_interval_sec,
    COUNT(DISTINCT member_id)                              AS captain_cnt
FROM user_ntile
GROUP BY dt,exp_group,ntile_bucket
ORDER BY dt,exp_group,ntile_bucket;

    ROUND(MAX(interval_sec),2)                              AS max_interval_sec,
    COUNT(DISTINCT member_id)                               AS user_cnt
FROM user_ntile
GROUP BY dt,exp_group,user_role,ntile_bucket
ORDER BY dt,exp_group,user_role,ntile_bucket;

WITH
-- 步骤1a：实验组首屏首次曝光（按 dt+user+role 粒度）
exp_first_screen AS (
    SELECT
        dt,
        CAST(member_id AS VARCHAR)                                          AS member_id,
        '实验组'                                                             AS exp_group,
        CASE WHEN COALESCE(JSON_EXTRACT_SCALAR(page_param,'$.groupid'),'')=''
             THEN '团长' ELSE '团员' END                                     AS user_role,
        MIN(log_timestamp)                                                   AS first_screen_ts
    FROM dw.dw_tfc_elmt_eps_ug_di
    WHERE dt BETWEEN '20260526' AND '20260602'
      AND p_game_tp='106' AND site_tp='shein' AND game_tp=106 AND page_tp='2'
      AND event_nm='expose_popup_selectattr'
      AND JSON_EXTRACT_SCALAR(activity_param,'$.popup_type')='groupbuycountdown_begin'
      AND JSON_EXTRACT_SCALAR(activity_param,'$.groupbuycountdown') IN ('A','B')
      AND member_id IS NOT NULL AND member_id > 0
    GROUP BY dt, CAST(member_id AS VARCHAR),
             CASE WHEN COALESCE(JSON_EXTRACT_SCALAR(page_param,'$.groupid'),'')=''
                  THEN '团长' ELSE '团员' END
),
-- 步骤1b：对照组首屏首次曝光（按 dt+user+role 粒度）
ctrl_first_screen AS (
    SELECT
        dt, CAST(member_id AS VARCHAR) AS member_id, '对照组' AS exp_group,
        CASE WHEN COALESCE(JSON_EXTRACT_SCALAR(page_param,'$.groupid'),'')=''
             THEN '团长' ELSE '团员' END AS user_role,
        MIN(log_timestamp) AS first_screen_ts
    FROM dw.dw_tfc_elmt_eps_ug_di
    WHERE dt BETWEEN '20260526' AND '20260602'
      AND p_game_tp='106' AND site_tp='shein' AND game_tp=106 AND page_tp='2'
      AND event_nm='expose_popup_selectattr'
      AND JSON_EXTRACT_SCALAR(activity_param,'$.popup_type')='groupbuyfreeshipping'
      AND member_id IS NOT NULL AND member_id > 0
    GROUP BY dt, CAST(member_id AS VARCHAR),
             CASE WHEN COALESCE(JSON_EXTRACT_SCALAR(page_param,'$.groupid'),'')=''
                  THEN '团长' ELSE '团员' END
),
-- 合并两组
all_first_screen AS (
    SELECT dt,member_id,exp_group,user_role,first_screen_ts FROM exp_first_screen
    UNION ALL
    SELECT dt,member_id,exp_group,user_role,first_screen_ts FROM ctrl_first_screen
),
-- 步骤2：立即购曝光明细（含 user_role 供角色配对）
orderdrawer_detail AS (
    SELECT
        dt, CAST(member_id AS VARCHAR) AS member_id,
        CASE WHEN COALESCE(JSON_EXTRACT_SCALAR(page_param,'$.groupid'),'')=''
             THEN '团长' ELSE '团员' END AS user_role,
        log_timestamp AS orderdrawer_ts
    FROM dw.dw_tfc_elmt_eps_ug_di
    WHERE dt BETWEEN '20260526' AND '20260602'
      AND p_game_tp='106' AND site_tp='shein' AND game_tp=106 AND page_tp='2'
      AND event_nm='expose_popup_selectattr'
      AND JSON_EXTRACT_SCALAR(activity_param,'$.popup_type')='orderdrawer'
      AND member_id IS NOT NULL AND member_id > 0
),
-- 步骤3：计算当天时间间隔（秒）· 粒度：dt×exp_group×user_role×member_id
user_interval AS (
    SELECT f.dt, f.exp_group, f.user_role, f.member_id,
           CAST(MIN(o.orderdrawer_ts)-f.first_screen_ts AS DOUBLE)/1000.0 AS interval_sec
    FROM all_first_screen f
    INNER JOIN orderdrawer_detail o
        ON f.dt=o.dt AND f.member_id=o.member_id
        AND f.user_role=o.user_role
        AND o.orderdrawer_ts > f.first_screen_ts
    GROUP BY f.dt,f.exp_group,f.user_role,f.member_id,f.first_screen_ts
),
-- 步骤4：跨日整体分桶（PARTITION 去掉dt，汇总后按组×角色独立分桶）
user_ntile AS (
    SELECT dt, exp_group, user_role, member_id, interval_sec,
           NTILE(10) OVER(
               PARTITION BY exp_group, user_role  -- ← 去掉dt，跨日整体排序
               ORDER BY interval_sec ASC
           ) AS ntile_bucket
    FROM user_interval
)
-- 步骤5：跨日汇总输出（无dt维度）
SELECT
    exp_group,
    user_role,
    CONCAT(CAST(ntile_bucket*10 AS VARCHAR),'分位')          AS percentile_group,
    ROUND(AVG(interval_sec),2)                               AS avg_interval_sec,
    ROUND(APPROX_PERCENTILE(interval_sec,0.5),2)             AS median_interval_sec,
    ROUND(MIN(interval_sec),2)                               AS min_interval_sec,
    ROUND(MAX(interval_sec),2)                               AS max_interval_sec,
    COUNT(DISTINCT CONCAT(dt,'_',member_id))                 AS user_cnt  -- 用户+天去重
FROM user_ntile
GROUP BY exp_group, user_role, ntile_bucket
ORDER BY exp_group, user_role, ntile_bucket;
