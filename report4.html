    dt, exp_group,

    GROUP BY dt, CAST(member_id AS VARCHAR),
             CASE WHEN COALESCE(JSON_EXTRACT_SCALAR(page_param,'$.groupid'),'')=''
                  THEN '团长' ELSE '团员' END
),
-- 步骤1b：对照组首屏首次曝光（按 dt+user+role 粒度）
ctrl_first_screen AS (
    SELECT
        dt, CAST(member_id AS VARCHAR) AS member_id, '对照组' AS exp_group,
        CASE WHEN COALESCE(JSON_EXTRACT_SCALAR(page_param,'$.groupid'),'')=''
             THEN '团长' ELSE '团员' END AS user_role,
        MIN(log_timestamp) AS first_screen_ts
    FROM dw.dw_tfc_elmt_eps_ug_di
    WHERE dt BETWEEN '20260526' AND '20260602'
      AND p_game_tp='106' AND site_tp='shein' AND game_tp=106 AND page_tp='2'
      AND event_nm='expose_popup_selectattr'
      AND JSON_EXTRACT_SCALAR(activity_param,'$.popup_type')='groupbuyfreeshipping'
      AND member_id IS NOT NULL AND member_id > 0
    GROUP BY dt, CAST(member_id AS VARCHAR),
             CASE WHEN COALESCE(JSON_EXTRACT_SCALAR(page_param,'$.groupid'),'')=''
                  THEN '团长' ELSE '团员' END
),
all_first_screen AS (
    SELECT dt,member_id,exp_group,user_role,first_screen_ts FROM exp_first_screen
    UNION ALL
    SELECT dt,member_id,exp_group,user_role,first_screen_ts FROM ctrl_first_screen
),
-- 步骤2：立即购曝光明细（含 user_role 供角色配对）
orderdrawer_detail AS (
    SELECT
        dt, CAST(member_id AS VARCHAR) AS member_id,
        CASE WHEN COALESCE(JSON_EXTRACT_SCALAR(page_param,'$.groupid'),'')=''
             THEN '团长' ELSE '团员' END AS user_role,
        log_timestamp AS orderdrawer_ts
    FROM dw.dw_tfc_elmt_eps_ug_di
    WHERE dt BETWEEN '20260526' AND '20260602'
      AND p_game_tp='106' AND site_tp='shein' AND game_tp=106 AND page_tp='2'
      AND event_nm='expose_popup_selectattr'
      AND JSON_EXTRACT_SCALAR(activity_param,'$.popup_type')='orderdrawer'
      AND member_id IS NOT NULL AND member_id > 0
),
-- 步骤3：计算当天时间间隔（秒）· 粒度：dt×exp_group×user_role×member_id
user_interval AS (
    SELECT f.dt, f.exp_group, f.user_role, f.member_id,
           CAST(MIN(o.orderdrawer_ts)-f.first_screen_ts AS DOUBLE)/1000.0 AS interval_sec
    FROM all_first_screen f
    INNER JOIN orderdrawer_detail o
        ON f.dt=o.dt AND f.member_id=o.member_id
        AND f.user_role=o.user_role
        AND o.orderdrawer_ts > f.first_screen_ts
    GROUP BY f.dt,f.exp_group,f.user_role,f.member_id,f.first_screen_ts
),
-- 步骤4：跨日整体分桶，PARTITION 去掉dt，按组×角色独立分桶
user_ntile AS (
    SELECT dt, exp_group, user_role, member_id, interval_sec,
           NTILE(10) OVER(
               PARTITION BY exp_group, user_role  -- ✓ 去掉dt，跨日整体排序
               ORDER BY interval_sec ASC
           ) AS ntile_bucket
    FROM user_interval
)
-- 步骤5：跨日汇总输出（去日期维度）
SELECT
    exp_group,
    user_role,
    CONCAT(CAST(ntile_bucket*10 AS VARCHAR),'分位')          AS percentile_group,
    ROUND(AVG(interval_sec),2)                               AS avg_interval_sec,
    ROUND(APPROX_PERCENTILE(interval_sec,0.5),2)             AS median_interval_sec,
    ROUND(MIN(interval_sec),2)                               AS min_interval_sec,
    ROUND(MAX(interval_sec),2)                               AS max_interval_sec,
    COUNT(DISTINCT CONCAT(dt,'_',member_id))                 AS user_cnt  -- 用户+天去重
FROM user_ntile
GROUP BY exp_group, user_role, ntile_bucket
ORDER BY exp_group, user_role, ntile_bucket;

-- =============================================
-- 需求：拼团游戏AB实验-团长从首屏首次曝光到立即购页面首次曝光的间隔时间分布（10分位分组）
-- 业务线：裂变拉新业务线
-- 生成时间：2026-06-04
-- 数据范围：20260526 ~ 20260603
-- 统计粒度：日
-- 关键说明：
-- 1. 指标口径：时间间隔 = (orderdrawer弹窗首次曝光时间戳 - 首屏首次曝光时间戳) / 1000，单位秒
--             实验组：popup_type='groupbuycountdown_begin' 且 groupbuycountdown IN ('A','B')
--             对照组：popup_type='groupbuyfreeshipping'
-- 2. 数据过滤：game_tp=106(拼团游戏)，page_tp='2'，site_tp='shein'，团长（groupid为空）
-- 3. 业务影响：2026-05-27 US站点首单数预计受影响30%-40%；KR/JP站点有营销活动变更
-- 4. 性能提示：dw_tfc_elmt_eps_ug_di 为用增专属天级曝光表（优于通用小时级 _hi 表），
--             已加 dt+p_game_tp+game_tp+page_tp+activity_nm 多维分区/字段过滤
-- 选表依据：dw.dw_tfc_elmt_eps_ug_di 优于 dw.dw_tfc_elmt_eps_hi，
--          原因：用增业务专属表、天级(_di)优于小时级(_hi)、无需 func_channel 分区
-- =============================================

WITH
-- ====================
-- Step 1: 获取首屏首次曝光（实验组+对照组，仅团长）
--   实验组：popup_type='groupbuycountdown_begin' 且 groupbuycountdown IN ('A','B')
--   对照组：popup_type='groupbuyfreeshipping'
--   团长识别：page_param 中 $.groupid 为空或空值
-- ====================
first_expose_base AS (
    SELECT
        dt                                                                          AS dt,
        CAST(member_id AS STRING)                                                   AS member_id,
        CASE
            WHEN get_json_object(activity_param, '$.popup_type') = 'groupbuycountdown_begin' THEN '实验组'
            ELSE '对照组'
        END                                                                         AS exp_group,
        MIN(log_timestamp)                                                          AS first_expose_ts
    FROM dw.dw_tfc_elmt_eps_ug_di
    WHERE dt BETWEEN '20260526' AND '20260603'
      AND p_game_tp = '106'                                                         -- 分区裁剪，提升性能
      AND game_tp IN (106)                                                          -- 拼团游戏
      AND page_tp = '2'                                                             -- 拼团主页/列表页
      AND activity_nm = 'expose_popup_selectattr'                                   -- 弹窗曝光事件
      AND site_tp = 'shein'
      AND COALESCE(get_json_object(page_param, '$.groupid'), '') = ''               -- 团长：groupid为空
      AND (
          (   get_json_object(activity_param, '$.popup_type') = 'groupbuycountdown_begin'
           AND get_json_object(activity_param, '$.groupbuycountdown') IN ('A', 'B') -- 实验组 A 或 B
          )
          OR get_json_object(activity_param, '$.popup_type') = 'groupbuyfreeshipping' -- 对照组
      )
    GROUP BY
        dt,
        CAST(member_id AS STRING),
        CASE
            WHEN get_json_object(activity_param, '$.popup_type') = 'groupbuycountdown_begin' THEN '实验组'
            ELSE '对照组'
        END
),

-- ====================
-- Step 2: 获取立即购页面首次曝光（仅团长）
--   popup_type='orderdrawer' 即立即购弹窗
-- ====================
order_expose_base AS (
    SELECT
        dt                                      AS dt,
        CAST(member_id AS STRING)               AS member_id,
        MIN(log_timestamp)                      AS order_expose_ts
    FROM dw.dw_tfc_elmt_eps_ug_di
    WHERE dt BETWEEN '20260526' AND '20260603'
      AND p_game_tp = '106'
      AND game_tp IN (106)
      AND page_tp = '2'
      AND activity_nm = 'expose_popup_selectattr'
      AND site_tp = 'shein'
      AND COALESCE(get_json_object(page_param, '$.groupid'), '') = ''               -- 团长
      AND get_json_object(activity_param, '$.popup_type') = 'orderdrawer'          -- 立即购弹窗
    GROUP BY
        dt,
        CAST(member_id AS STRING)
),

-- ====================
-- Step 3: 计算时间间隔（秒），仅保留立即购在首屏曝光之后的有效记录
-- ====================
interval_base AS (
    SELECT
        fe.dt                                                                       AS dt,
        fe.member_id                                                                AS member_id,
        fe.exp_group                                                                AS exp_group,
        ROUND((oe.order_expose_ts - fe.first_expose_ts) / 1000.0, 2)              AS interval_sec
    FROM first_expose_base fe
    INNER JOIN order_expose_base oe
        ON fe.dt = oe.dt
        AND fe.member_id = oe.member_id
    WHERE oe.order_expose_ts >= fe.first_expose_ts                                  -- 时序过滤：立即购须在首屏之后
),

-- ====================
-- Step 4: NTILE(10) 分位分组（按每日每实验组单独计算，升序排列）
-- ====================
ntile_base AS (
    SELECT
        dt,
        member_id,
        exp_group,
        interval_sec,
        NTILE(10) OVER (
            PARTITION BY dt, exp_group
            ORDER BY interval_sec ASC
        )                                                                           AS ntile_bucket
    FROM interval_base
),

-- ====================
-- Step 5: 从拼团成员快照表取团长拼团记录
--   全量快照表，取最新分区 dt='2026060323'，通过 create_time 过滤统计日期范围
-- ====================
captain_group_base AS (
    SELECT
        REPLACE(SUBSTR(create_time, 1, 10), '-', '')    AS order_dt,
        CAST(user_id AS STRING)                          AS member_id,
        group_buy_id,
        bill_no,
        group_status
    FROM dw.dw_mkt_group_buy_member_h
    WHERE dt = '2026060323'                                                         -- 全量快照最新分区
      AND member_role = 1                                                            -- 仅团长
      AND LOWER(COALESCE(act_name_en, '')) NOT LIKE '%test%'                        -- 过滤测试活动
      AND REPLACE(SUBSTR(create_time, 1, 10), '-', '') BETWEEN '20260526' AND '20260603'
),

-- ====================
-- Step 6: 从拼团成员快照表取团员拼团记录
-- ====================
member_group_base AS (
    SELECT
        group_buy_id,
        bill_no
    FROM dw.dw_mkt_group_buy_member_h
    WHERE dt = '2026060323'
      AND member_role = 2                                                            -- 仅团员
      AND LOWER(COALESCE(act_name_en, '')) NOT LIKE '%test%'
      AND REPLACE(SUBSTR(create_time, 1, 10), '-', '') BETWEEN '20260526' AND '20260603'
),

-- ====================
-- Step 7: 时间间隔统计量（均值、中位数、最值、团长数）
-- ====================
interval_stats AS (
    SELECT
        dt,
        exp_group,
        ntile_bucket,
        ROUND(AVG(interval_sec), 2)                       AS avg_interval_sec,
        ROUND(PERCENTILE_APPROX(interval_sec, 0.5), 2)    AS median_interval_sec,
        ROUND(MIN(interval_sec), 2)                       AS min_interval_sec,
        ROUND(MAX(interval_sec), 2)                       AS max_interval_sec,
        COUNT(DISTINCT member_id)                         AS captain_cnt
    FROM ntile_base
    GROUP BY dt, exp_group, ntile_bucket
),

-- ====================
-- Step 8: 团长拼团支付订单数 & 团长拼团成功支付订单数
--   支付订单：表中所有订单，不限制 bill_status
--   成功支付订单：group_status=2（已成团）
-- ====================
captain_order_stats AS (
    SELECT
        nb.dt                                                                       AS dt,
        nb.exp_group                                                                AS exp_group,
        nb.ntile_bucket                                                             AS ntile_bucket,
        COUNT(DISTINCT cg.bill_no)                                                  AS captain_order_cnt,
        COUNT(DISTINCT CASE WHEN cg.group_status = 2 THEN cg.bill_no ELSE NULL END) AS captain_succ_order_cnt
    FROM ntile_base nb
    LEFT JOIN captain_group_base cg
        ON nb.dt = cg.order_dt
        AND nb.member_id = cg.member_id
    GROUP BY nb.dt, nb.exp_group, nb.ntile_bucket
),

-- ====================
-- Step 9: 团员拼团支付订单数
--   通过团长所在拼团（group_buy_id）关联团员，统计团员订单
-- ====================
member_order_stats AS (
    SELECT
        nb.dt                                                                       AS dt,
        nb.exp_group                                                                AS exp_group,
        nb.ntile_bucket                                                             AS ntile_bucket,
        COUNT(DISTINCT mg.bill_no)                                                  AS member_order_cnt
    FROM ntile_base nb
    INNER JOIN captain_group_base cg
        ON nb.dt = cg.order_dt
        AND nb.member_id = cg.member_id
    INNER JOIN member_group_base mg
        ON cg.group_buy_id = mg.group_buy_id
    GROUP BY nb.dt, nb.exp_group, nb.ntile_bucket
)

-- ====================
-- 最终结果：按日期、实验组别、10分位段输出
-- ====================
SELECT
    ist.dt                                                                          AS dt,                          -- 日期
    ist.exp_group                                                                   AS exp_group,                   -- 实验组别（对照组/实验组）
    CONCAT(CAST(ist.ntile_bucket * 10 AS STRING), '分位')                           AS percentile_group,            -- 10分位段（10分位~100分位）
    ist.avg_interval_sec                                                            AS avg_interval_sec,            -- 时间间隔均值（秒）
    ist.median_interval_sec                                                         AS median_interval_sec,         -- 时间间隔中位数（秒）
    ist.min_interval_sec                                                            AS min_interval_sec,            -- 时间间隔最小值（秒）
    ist.max_interval_sec                                                            AS max_interval_sec,            -- 时间间隔最大值（秒）
    ist.captain_cnt                                                                 AS captain_cnt,                 -- 团长数
    COALESCE(cos.captain_order_cnt, 0)                                              AS captain_order_cnt,           -- 团长拼团支付订单数
    COALESCE(mos.member_order_cnt, 0)                                               AS member_order_cnt,            -- 团员拼团支付订单数
    COALESCE(cos.captain_succ_order_cnt, 0)                                         AS captain_succ_order_cnt       -- 团长拼团成功支付订单数
FROM interval_stats ist
LEFT JOIN captain_order_stats cos
    ON ist.dt = cos.dt
    AND ist.exp_group = cos.exp_group
    AND ist.ntile_bucket = cos.ntile_bucket
LEFT JOIN member_order_stats mos
    ON ist.dt = mos.dt
    AND ist.exp_group = mos.exp_group
    AND ist.ntile_bucket = mos.ntile_bucket
ORDER BY
    ist.dt          ASC,
    ist.exp_group   ASC,
    ist.ntile_bucket ASC
;
ENGINE: Trino SQL · STRING→VARCHAR · GET_JSON_OBJECT→JSON_EXTRACT_SCALAR · PERCENTILE_APPROX→APPROX_PERCENTILE 
-- =============================================
-- 需求：拼团游戏AB实验-团长从首屏首次曝光到立即购页面首次曝光的间隔时间分布（10分位分组）
-- 业务线：裂变拉新业务线
-- 生成时间：2026-06-04
-- 数据范围：20260526 ~ 20260603
-- 统计粒度：日
-- 执行引擎：Trino SQL
-- 关键说明：
-- 1. 指标口径：时间间隔 = (orderdrawer弹窗首次曝光时间戳 - 首屏首次曝光时间戳) / 1000，单位秒
--             实验组：popup_type='groupbuycountdown_begin' 且 groupbuycountdown IN ('A','B')
--             对照组：popup_type='groupbuyfreeshipping'
-- 2. 数据过滤：game_tp=106(拼团游戏)，page_tp='2'，site_tp='shein'，团长（groupid为空）
-- 3. 业务影响：2026-05-27 US站点首单数预计受影响30%-40%；KR/JP站点有营销活动变更
-- 4. 性能提示：dw_tfc_elmt_eps_ug_di 为用增专属天级曝光表（优于通用小时级 _hi 表），
--             已加 dt+p_game_tp+game_tp+page_tp+activity_nm 多维分区/字段过滤
-- =============================================

WITH
-- ====================
-- Step 1: 获取首屏首次曝光（实验组+对照组，仅团长）
--   实验组：popup_type='groupbuycountdown_begin' 且 groupbuycountdown IN ('A','B')
--   对照组：popup_type='groupbuyfreeshipping'
--   团长识别：page_param 中 $.groupid 为空或空值
-- ====================
first_expose_base AS (
    SELECT
        dt                                                                          AS dt,
        CAST(member_id AS VARCHAR)                                                  AS member_id,
        CASE
            WHEN JSON_EXTRACT_SCALAR(activity_param, '$.popup_type') = 'groupbuycountdown_begin' THEN '实验组'
            ELSE '对照组'
        END                                                                         AS exp_group,
        MIN(log_timestamp)                                                          AS first_expose_ts
    FROM dw.dw_tfc_elmt_eps_ug_di
    WHERE dt BETWEEN '20260526' AND '20260603'
      AND p_game_tp = '106'                                                         -- 分区裁剪，提升性能
      AND game_tp IN (106)                                                          -- 拼团游戏
      AND page_tp = '2'                                                             -- 拼团主页/列表页
      AND activity_nm = 'expose_popup_selectattr'                                   -- 弹窗曝光事件
      AND site_tp = 'shein'
      AND COALESCE(JSON_EXTRACT_SCALAR(page_param, '$.groupid'), '') = ''           -- 团长：groupid为空
      AND (
          (   JSON_EXTRACT_SCALAR(activity_param, '$.popup_type') = 'groupbuycountdown_begin'
           AND JSON_EXTRACT_SCALAR(activity_param, '$.groupbuycountdown') IN ('A', 'B')
          )
          OR JSON_EXTRACT_SCALAR(activity_param, '$.popup_type') = 'groupbuyfreeshipping'
      )
    GROUP BY
        dt,
        CAST(member_id AS VARCHAR),
        CASE
            WHEN JSON_EXTRACT_SCALAR(activity_param, '$.popup_type') = 'groupbuycountdown_begin' THEN '实验组'
            ELSE '对照组'
        END
),

-- ====================
-- Step 2: 获取立即购页面首次曝光（仅团长）
--   popup_type='orderdrawer' 即立即购弹窗
-- ====================
order_expose_base AS (
    SELECT
        dt                                      AS dt,
        CAST(member_id AS VARCHAR)              AS member_id,
        MIN(log_timestamp)                      AS order_expose_ts
    FROM dw.dw_tfc_elmt_eps_ug_di
    WHERE dt BETWEEN '20260526' AND '20260603'
      AND p_game_tp = '106'
      AND game_tp IN (106)
      AND page_tp = '2'
      AND activity_nm = 'expose_popup_selectattr'
      AND site_tp = 'shein'
      AND COALESCE(JSON_EXTRACT_SCALAR(page_param, '$.groupid'), '') = ''           -- 团长
      AND JSON_EXTRACT_SCALAR(activity_param, '$.popup_type') = 'orderdrawer'      -- 立即购弹窗
    GROUP BY
        dt,
        CAST(member_id AS VARCHAR)
),

-- ====================
-- Step 3: 计算时间间隔（秒），仅保留立即购在首屏曝光之后的有效记录
-- ====================
interval_base AS (
    SELECT
        fe.dt                                                                       AS dt,
        fe.member_id                                                                AS member_id,
        fe.exp_group                                                                AS exp_group,
        ROUND((oe.order_expose_ts - fe.first_expose_ts) / 1000.0, 2)              AS interval_sec
    FROM first_expose_base fe
    INNER JOIN order_expose_base oe
        ON fe.dt = oe.dt
        AND fe.member_id = oe.member_id
    WHERE oe.order_expose_ts >= fe.first_expose_ts                                  -- 时序过滤：立即购须在首屏之后
),

-- ====================
-- Step 4: NTILE(10) 分位分组（按每日每实验组单独计算，升序排列）
-- ====================
ntile_base AS (
    SELECT
        dt,
        member_id,
        exp_group,
        interval_sec,
        NTILE(10) OVER (
            PARTITION BY dt, exp_group
            ORDER BY interval_sec ASC
        )                                                                           AS ntile_bucket
    FROM interval_base
),

-- ====================
-- Step 5: 从拼团成员快照表取团长拼团记录
--   全量快照表，取最新分区 dt='2026060323'，通过 create_time 过滤统计日期范围
-- ====================
captain_group_base AS (
    SELECT
        REPLACE(SUBSTR(create_time, 1, 10), '-', '')    AS order_dt,
        CAST(user_id AS VARCHAR)                         AS member_id,
        group_buy_id,
        bill_no,
        group_status
    FROM dw.dw_mkt_group_buy_member_h
    WHERE dt = '2026060323'                                                         -- 全量快照最新分区
      AND member_role = 1                                                            -- 仅团长
      AND LOWER(COALESCE(act_name_en, '')) NOT LIKE '%test%'                        -- 过滤测试活动
      AND REPLACE(SUBSTR(create_time, 1, 10), '-', '') BETWEEN '20260526' AND '20260603'
),

-- ====================
-- Step 6: 从拼团成员快照表取团员拼团记录
-- ====================
member_group_base AS (
    SELECT
        group_buy_id,
        bill_no
    FROM dw.dw_mkt_group_buy_member_h
    WHERE dt = '2026060323'
      AND member_role = 2                                                            -- 仅团员
      AND LOWER(COALESCE(act_name_en, '')) NOT LIKE '%test%'
      AND REPLACE(SUBSTR(create_time, 1, 10), '-', '') BETWEEN '20260526' AND '20260603'
),

-- ====================
-- Step 7: 时间间隔统计量（均值、中位数、最值、团长数）
-- ====================
interval_stats AS (
    SELECT
        dt,
        exp_group,
        ntile_bucket,
        ROUND(AVG(interval_sec), 2)                       AS avg_interval_sec,
        ROUND(APPROX_PERCENTILE(interval_sec, 0.5), 2)    AS median_interval_sec,
        ROUND(MIN(interval_sec), 2)                       AS min_interval_sec,
        ROUND(MAX(interval_sec), 2)                       AS max_interval_sec,
        COUNT(DISTINCT member_id)                         AS captain_cnt
    FROM ntile_base
    GROUP BY dt, exp_group, ntile_bucket
),

-- ====================
-- Step 8: 团长拼团支付订单数 & 团长拼团成功支付订单数
-- ====================
captain_order_stats AS (
    SELECT
        nb.dt                                                                       AS dt,
        nb.exp_group                                                                AS exp_group,
        nb.ntile_bucket                                                             AS ntile_bucket,
        COUNT(DISTINCT cg.bill_no)                                                  AS captain_order_cnt,
        COUNT(DISTINCT CASE WHEN cg.group_status = 2 THEN cg.bill_no ELSE NULL END) AS captain_succ_order_cnt
    FROM ntile_base nb
    LEFT JOIN captain_group_base cg
        ON nb.dt = cg.order_dt
        AND nb.member_id = cg.member_id
    GROUP BY nb.dt, nb.exp_group, nb.ntile_bucket
),

-- ====================
-- Step 9: 团员拼团支付订单数
-- ====================
member_order_stats AS (
    SELECT
        nb.dt                                                                       AS dt,
        nb.exp_group                                                                AS exp_group,
        nb.ntile_bucket                                                             AS ntile_bucket,
        COUNT(DISTINCT mg.bill_no)                                                  AS member_order_cnt
    FROM ntile_base nb
    INNER JOIN captain_group_base cg
        ON nb.dt = cg.order_dt
        AND nb.member_id = cg.member_id
    INNER JOIN member_group_base mg
        ON cg.group_buy_id = mg.group_buy_id
    GROUP BY nb.dt, nb.exp_group, nb.ntile_bucket
)

-- ====================
-- 最终结果：按日期、实验组别、10分位段输出
-- ====================
SELECT
    ist.dt                                                                          AS dt,                          -- 日期
    ist.exp_group                                                                   AS exp_group,                   -- 实验组别（对照组/实验组）
    CONCAT(CAST(ist.ntile_bucket * 10 AS VARCHAR), '分位')                          AS percentile_group,            -- 10分位段（10分位~100分位）
    ist.avg_interval_sec                                                            AS avg_interval_sec,            -- 时间间隔均值（秒）
    ist.median_interval_sec                                                         AS median_interval_sec,         -- 时间间隔中位数（秒）
    ist.min_interval_sec                                                            AS min_interval_sec,            -- 时间间隔最小值（秒）
    ist.max_interval_sec                                                            AS max_interval_sec,            -- 时间间隔最大值（秒）
    ist.captain_cnt                                                                 AS captain_cnt,                 -- 团长数
    COALESCE(cos.captain_order_cnt, 0)                                              AS captain_order_cnt,           -- 团长拼团支付订单数
    COALESCE(mos.member_order_cnt, 0)                                               AS member_order_cnt,            -- 团员拼团支付订单数
    COALESCE(cos.captain_succ_order_cnt, 0)                                         AS captain_succ_order_cnt       -- 团长拼团成功支付订单数
FROM interval_stats ist
LEFT JOIN captain_order_stats cos
    ON ist.dt = cos.dt
    AND ist.exp_group = cos.exp_group
    AND ist.ntile_bucket = cos.ntile_bucket
LEFT JOIN member_order_stats mos
    ON ist.dt = mos.dt
    AND ist.exp_group = mos.exp_group
    AND ist.ntile_bucket = mos.ntile_bucket
ORDER BY
    ist.dt          ASC,
    ist.exp_group   ASC,
    ist.ntile_bucket ASC
;
