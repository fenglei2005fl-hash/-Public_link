ENGINE: Hive · 2026-05-26~2026-06-01 · 日粒度 
WITH
-- Step 1: 圈定近7天开团团长（时间基准）
leader_base AS (
    SELECT
         REPLACE(SUBSTR(create_time, 1, 10), '-', '')  AS open_dt
        ,CAST(user_id AS BIGINT)                        AS leader_user_id
        ,group_buy_id
        ,bill_no                                        AS leader_bill_no
    FROM dw.dw_mkt_group_buy_member_h
    WHERE dt = '2026060123'
      AND member_role = 1
      AND LOWER(COALESCE(act_name_en, '')) NOT LIKE '%test%'
      AND REPLACE(SUBSTR(create_time, 1, 10), '-', '') >= '20260526'
      AND REPLACE(SUBSTR(create_time, 1, 10), '-', '') <= '20260601'
),
-- Step 2: 全量团员（时间归因到其所在团的团长开团日）
member_raw AS (
    SELECT CAST(user_id AS BIGINT) AS member_user_id, group_buy_id, bill_no AS member_bill_no
    FROM dw.dw_mkt_group_buy_member_h
    WHERE dt = '2026060123' AND member_role = 2
),
-- Step 3: 团员挂载到团长开团日期
member_base AS (
    SELECT l.open_dt, m.member_user_id, m.member_bill_no
    FROM leader_base l LEFT JOIN member_raw m ON l.group_buy_id = m.group_buy_id
),
-- Step 4: 团长维度聚合
leader_stats AS (
    SELECT open_dt,
           COUNT(DISTINCT leader_user_id) AS open_user_cnt,
           COUNT(DISTINCT leader_bill_no) AS leader_order_cnt
    FROM leader_base GROUP BY open_dt
),
-- Step 5: 团员维度聚合
member_stats AS (
    SELECT open_dt,
           COUNT(DISTINCT member_user_id) AS join_user_cnt,
           COUNT(DISTINCT member_bill_no) AS member_order_cnt
    FROM member_base GROUP BY open_dt
)
-- 最终输出
SELECT
     l.open_dt
    ,l.open_user_cnt                                    -- 开团用户数
    ,COALESCE(m.join_user_cnt,    0) AS join_user_cnt   -- 参团用户数
    ,l.leader_order_cnt                                 -- 团长支付订单
    ,COALESCE(m.member_order_cnt, 0) AS member_order_cnt -- 团员支付订单
FROM leader_stats l LEFT JOIN member_stats m ON l.open_dt = m.open_dt
ORDER BY l.open_dt ASC;
说明项内容
时间归因4 个指标全部归因到团长 create_time 日期，数据字典明确规定「默认以团长开团时间为准」
支付口径COUNT(DISTINCT bill_no)，NULL 自动排除，不限制 bill_status（依数据字典规则）
全量分区取 dt='2026060123' 单分区，覆盖所有历史数据，无需多分区扫描

ENGINE: Hive · game_type=106 · 同日漏斗 · 2026-05-26~2026-06-01 
WITH
-- Step 1 基准: 访问拼团主页的已登录用户
vst_base AS (
    SELECT dt AS open_dt, CAST(member_id AS BIGINT) AS member_id
    FROM dw.dw_ssa_ug_game_page_vst_di
    WHERE dt >= '20260526' AND dt <= '20260601'
      AND game_type = 106
      AND page_type IN ('2')           -- 拼团主页/列表页
      AND site_tp = 'shein'
      AND login_flag = '1'             -- 仅登录用户
      AND env = 'production'
    GROUP BY dt, CAST(member_id AS BIGINT)
),
vst_cnt AS (SELECT open_dt, COUNT(DISTINCT member_id) AS vst_uv FROM vst_base GROUP BY open_dt),

-- Step 2: 访问用户中有商品点击的用户（button_type='250'）
click_base AS (
    SELECT c.dt AS open_dt, CAST(c.member_id AS BIGINT) AS member_id
    FROM dw.dw_ssa_ug_game_elmt_clk_di c
    INNER JOIN vst_base v ON CAST(c.member_id AS BIGINT) = v.member_id AND c.dt = v.open_dt
    WHERE c.dt >= '20260526' AND c.dt <= '20260601'
      AND c.game_type IN (106,'106') AND c.page_type IN ('2')
      AND c.activity_nm = 'click_game_button' AND c.button_type = '250'
      AND c.site_tp = 'shein'
    GROUP BY c.dt, CAST(c.member_id AS BIGINT)
),
click_cnt AS (SELECT open_dt, COUNT(DISTINCT member_id) AS click_uv FROM click_base GROUP BY open_dt),

-- Step 3: 商品点击用户中有加购行为的用户（button_type='50'）
cart_base AS (
    SELECT c.dt AS open_dt, CAST(c.member_id AS BIGINT) AS member_id
    FROM dw.dw_ssa_ug_game_elmt_clk_di c
    INNER JOIN click_base cl ON CAST(c.member_id AS BIGINT) = cl.member_id AND c.dt = cl.open_dt
    WHERE c.dt >= '20260526' AND c.dt <= '20260601'
      AND c.game_type IN (106,'106') AND c.button_type = '50'
      AND c.activity_nm = 'click_game_button' AND c.site_tp = 'shein'
    GROUP BY c.dt, CAST(c.member_id AS BIGINT)
),
cart_cnt AS (SELECT open_dt, COUNT(DISTINCT member_id) AS cart_uv FROM cart_base GROUP BY open_dt),

-- Step 4: 加购用户中完成支付开团的团长
pay_base AS (
    SELECT cb.open_dt, cb.member_id FROM cart_base cb
    INNER JOIN (
        SELECT CAST(user_id AS BIGINT) AS member_id,
               REPLACE(SUBSTR(create_time, 1, 10), '-', '') AS pay_dt
        FROM dw.dw_mkt_group_buy_member_h
        WHERE dt = '2026060123' AND member_role = 1 AND bill_no IS NOT NULL
          AND LOWER(COALESCE(act_name_en, '')) NOT LIKE '%test%'
          AND REPLACE(SUBSTR(create_time, 1, 10), '-', '') >= '20260526'
          AND REPLACE(SUBSTR(create_time, 1, 10), '-', '') <= '20260601'
    ) pay ON cb.member_id = pay.member_id AND cb.open_dt = pay.pay_dt
),
pay_cnt AS (SELECT open_dt, COUNT(DISTINCT member_id) AS pay_uv FROM pay_base GROUP BY open_dt)

-- 最终输出: 各步骤 UV + 逐层转化率
SELECT
     vc.open_dt
    ,vc.vst_uv                                                               -- Step1 访问 UV
    ,COALESCE(cc.click_uv, 0)                                                AS click_uv
    ,COALESCE(ca.cart_uv,  0)                                                AS cart_uv
    ,COALESCE(pc.pay_uv,   0)                                                AS pay_uv
    ,ROUND(COALESCE(cc.click_uv,0) / NULLIF(vc.vst_uv,   0) * 100, 2)      AS click_cvr   -- 访问→点击 %
    ,ROUND(COALESCE(ca.cart_uv, 0) / NULLIF(cc.click_uv, 0) * 100, 2)      AS cart_cvr    -- 点击→加购 %
    ,ROUND(COALESCE(pc.pay_uv,  0) / NULLIF(ca.cart_uv,  0) * 100, 2)      AS pay_cvr     -- 加购→支付 %
    ,ROUND(COALESCE(pc.pay_uv,  0) / NULLIF(vc.vst_uv,   0) * 100, 2)      AS overall_cvr -- 总转化率 %
FROM vst_cnt vc
LEFT JOIN click_cnt cc ON vc.open_dt = cc.open_dt
LEFT JOIN cart_cnt  ca ON vc.open_dt = ca.open_dt
LEFT JOIN pay_cnt   pc ON vc.open_dt = pc.open_dt
ORDER BY vc.open_dt ASC;
说明项内容
漏斗口径严格 INNER JOIN 追踪同一批用户，每步只保留经历上一步的人，避免漏斗断层
时间窗口同日漏斗（访问/点击/加购/支付在同一 dt 内），如需跨日需修改 JOIN 条件
访问口径page_type='2' = 拼团主页/列表页（数据字典明确）
点击/加购点击 button_type='250'，加购 button_type='50'（数据字典明确）
支付口径取拼团快照表团长且 bill_no IS NOT NULL
